export interface Register {}
export type RegisteredSchemas = Register extends { schemas: infer S } ? S : Record<string, unknown>;
type SchemaKeys = Register extends { schemas: infer S } ? keyof S & string : never;
type SchemaTypeKeys = Register extends { schemaTypes: infer T } ? keyof T & string : never;
type ExtractID<Keys extends string, NS extends string> = Keys extends `${NS}:${infer ID}` ? ID : never;
type ValidKeys<T, DefaultNS extends string | undefined> = DefaultNS extends string ? (keyof T & string) | ExtractID<keyof T & string, DefaultNS> : keyof T & string;
type ResolveKey<K extends string, DefaultNS extends string | undefined> = K extends `${string}:${string}` ? K : DefaultNS extends string ? `${DefaultNS}:${K}` : K;
export type XSchemaType<N extends SchemaTypeKeys> = Register extends { schemaTypes: infer T } ? N extends keyof T ? T[N] : never : never;
export type XSchemaConfig<T extends Record<string, unknown> = RegisteredSchemas> = { schemas?: T; defaultNamespace?: string };
export function createXSchemaClient<const T extends Record<string, unknown> = RegisteredSchemas, const DefaultNS extends string | undefined = undefined>(config: XSchemaConfig<T> & { defaultNamespace?: DefaultNS }): <K extends ValidKeys<T, DefaultNS>>(key: K) => T[ResolveKey<K, DefaultNS> & keyof T];

import { z } from "zod"
const user_Profile = z.object({ id: z.string(), name: z.string().optional(), email: z.string().email() }).passthrough();
type user_ProfileType = z.infer<typeof user_Profile>;
const app_Config = z.object({ theme: z.enum(["light", "dark"]), language: z.string(), debug: z.boolean().optional() }).passthrough();
type app_ConfigType = z.infer<typeof app_Config>;
export const schemas = { "user:Profile": user_Profile, "app:Config": app_Config } as const;
export type SchemaTypes = { "user:Profile": user_ProfileType; "app:Config": app_ConfigType };

// @filename: index.ts
// ---cut---
import { createXSchemaClient } from "@xschemadev/client";

## Using the Client

The xschema client provides type-safe runtime schema lookups with full TypeScript autocompletion.

### createXSchemaClient

Creates a function for looking up schemas by key with compile-time type safety.

```typescript
import { createXSchemaClient } from "@xschemadev/client";
import { schemas } from "./.xschema/xschema.gen";

const xschema = createXSchemaClient({
  schemas,
  defaultNamespace: "user", // optional
});
```

#### Options

| Option | Type | Description |
|--------|------|-------------|
| `schemas` | `Record<string, unknown>` | The generated schemas object from `.xschema/xschema.gen` |
| `defaultNamespace` | `string` | Optional namespace to use when looking up schemas without a prefix |

#### Schema lookup

Look up schemas using the `namespace:id` format:

```typescript
// Full key format
const profileSchema = xschema("user:Profile");
const configSchema = xschema("app:Config");
```

With `defaultNamespace` set, you can omit the namespace for that namespace's schemas:

```typescript
const xschema = createXSchemaClient({
  schemas,
  defaultNamespace: "user",
});

// Shorthand - resolves to "user:Profile"
const profile = xschema("Profile");

// Still need full key for other namespaces
const config = xschema("app:Config");
```

### XSchemaType

Extract TypeScript types from your schemas by key:

```typescript
import type { XSchemaType } from "@xschemadev/client";

type Profile = XSchemaType<"user:Profile">;
type Config = XSchemaType<"app:Config">;
```

<Callout>
`XSchemaType` only accepts full `namespace:id` keys for explicitness. Use the client for shorthand lookups.
</Callout>

#### Using in function signatures

```typescript
import type { XSchemaType } from "@xschemadev/client";

type Profile = XSchemaType<"user:Profile">;

function createUser(data: Profile): void {
  console.log(data.email); // fully typed
}

function validateAndCreate(input: unknown): Profile {
  const schema = xschema("user:Profile");
  return schema.parse(input);
}
```

### Schema types explained

xschema adapters can generate three kinds of outputs:

#### Schema + Type (most common)

Adapters like Zod, ArkType, Effect, and Valibot generate both runtime validators and types. Both `xschema()` and `XSchemaType<>` work:

```typescript
const schema = xschema("user:Profile"); // runtime validator
type Profile = XSchemaType<"user:Profile">; // extracted type
```

#### Type-only

Some adapters may generate only types without runtime validators:

```typescript
// Only XSchemaType works
type Profile = XSchemaType<"user:Profile">;

// This would be a compile error - no runtime validator
// const schema = xschema("user:Profile");
```

#### Schema-only

Some adapters may generate validators that don't expose inferred types. `XSchemaType` returns `typeof` the schema:

```typescript
const schema = xschema("user:Profile");
type Profile = XSchemaType<"user:Profile">; // typeof schema
```

## What's Generated

The `.xschema/xschema.gen.ts` file contains:

- **Validators**: Native Zod schemas (or ArkType types, Effect schemas, etc.)
- **Types**: TypeScript types inferred from the validators
- **Schema registry**: An object mapping `namespace:id` to validators

Example output for a Zod adapter:

```typescript
// Generated by xschema - DO NOT EDIT
import { z } from "zod"

const user_Profile = z.object({
  id: z.string(),
  name: z.string().optional(),
  email: z.string().email(),
});
type user_ProfileType = z.infer<typeof user_Profile>;

export const schemas = {
  "user:Profile": user_Profile,
} as const;

export type SchemaTypes = {
  "user:Profile": user_ProfileType;
};
```
